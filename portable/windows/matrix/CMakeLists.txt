cmake_minimum_required(VERSION 3.20)
project(GlowMatrixWindows VERSION 2.0 LANGUAGES C)
if(NOT WIN32)
  message(FATAL_ERROR "This target is the native Windows VT adapter. Unix builds use the existing payload patch.")
endif()
if(NOT MSVC)
  message(FATAL_ERROR "Build the release with MSVC so /MT removes external C runtime DLL requirements.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(cmatrix-gm main.c unimatrix.c vt_terminal.c getopt_compat.c)
target_compile_definitions(cmatrix-gm PRIVATE _CRT_SECURE_NO_WARNINGS)
target_compile_options(cmatrix-gm PRIVATE /utf-8 /W4)
include(CTest)
if(BUILD_TESTING)
  add_executable(matrix-getopt-test test_getopt.c getopt_compat.c)
  add_test(NAME matrix-getopt COMMAND matrix-getopt-test)
  add_test(NAME matrix-help COMMAND cmatrix-gm --help)
endif()
add_custom_command(TARGET cmatrix-gm POST_BUILD
  COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:cmatrix-gm>" "$<TARGET_FILE_DIR:cmatrix-gm>/cmatrix-uni.exe"
  COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:cmatrix-gm>" "$<TARGET_FILE_DIR:cmatrix-gm>/cmatrix-uni-native.exe"
  VERBATIM)
